Code 
library(dplyr)

# attach true 2k watts (Workout 31) to each athlete
data.model <- data.no.hr |>
  group_by(Name) |>
  mutate(`2k.Watts` = Watts[Workout.Number == 31][1]) |>
  ungroup()

# fit global fatigue exponent
fit <- nls(
  `2k.Watts` ~ Watts * (Time..s. / 390)^d,
  data = data.model |> filter(Workout.Number != 31),
  start = list(d = 0.07)
)

# extract fitted exponent
d_hat <- coef(fit)["d"]

# compute predicted 2k watts for every workout
data.model <- data.model |>
  mutate(
    Predicted.2k = Watts * (Time..s. / 390)^d_hat
  )

# optional: check prediction stability by athlete
sd_table <- data.model |>
  group_by(Name) |>
  summarize(
    sd_pred = sd(Predicted.2k.new, na.rm = TRUE),
    mean_pred = mean(Predicted.2k.new, na.rm = TRUE)
  )
